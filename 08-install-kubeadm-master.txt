# 1.  Initialize Master (Stop swap first)
swapon -s
sudo swapoff -a
sudo kubeadm init --pod-network-cidr=10.244.XX.0/16
sudo kubeadm init --pod-network-cidr=10.244.0.0/16

# kubeadm join 192.168.100.114:6443 --token i3ue09.2d9tgbdlaot248mm --discovery-token-ca-cert-hash sha256:xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx

# 2.  Copy admin configuration
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config

# 3. Install POD Network Flannel (kube-flannel.yml)
wget https://raw.githubusercontent.com/coreos/flannel/master/Documentation/kube-flannel.yml
kubectl apply -f kube-flannel.yml
kubectl get pods --all-namespaces --watch

# 4. Verify config & cluster
kubectl config view
kubectl cluster-info

# 5. Show Token and token-ca-cert-hash
sudo kubeadm token list

## Create new token
sudo kubeadm token create --ttl 0h0m0s # never expire
sudo kubeadm token create --ttl 720h0m0s  # 30 days

## Retrive token-ca-cert-hash value on any of the control plane node within the cluster.
sudo openssl x509 -pubkey -in /etc/kubernetes/pki/ca.crt | openssl rsa -pubin -outform der 2>/dev/null | openssl dgst -sha256 -hex | sed 's/^.* //'


# 6. Remove node
kubectl drain kube-stack-kube-worker-instance-3hziikzbzdu2 --ignore-daemonsets --delete-local-data --delete-emptydir-data
## Drain already-deleted node
kubectl drain kube-stack-kube-worker-instance-3hziikzbzdu2 --ignore-daemonsets --delete-emptydir-data --grace-period=0
kubectl drain xxxx --ignore-daemonsets --delete-emptydir-data --grace-period=0
##Delete node
kubectl delete node kube-stack-kube-worker-instance-3hziikzbzdu2
kubectl get nodes

# 7. Reset
sudo kubeadm reset -y


# flush all chains
sudo iptables -P INPUT ACCEPT
sudo iptables -P FORWARD ACCEPT
sudo iptables -P OUTPUT ACCEPT
sudo iptables -F
sudo iptables -t nat -F
sudo iptables -t filter -F
sudo iptables -t mangle -F
# delete all chains
sudo iptables -X
